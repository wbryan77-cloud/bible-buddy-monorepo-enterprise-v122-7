// ---------- API: Activity / Dismisses CSV export ----------

function exportCsvFromArray(rows, res, name) {
  if (!Array.isArray(rows)) rows = [];
  const keys = [...new Set(rows.flatMap(r => Object.keys(r || {})))];
  const header = keys.join(',');
  const lines = rows.map(r =>
    keys
      .map(k => {
        const v = r && r[k] != null ? String(r[k]) : '';
        // naive CSV escape
        if (v.includes(',') || v.includes('"') || v.includes('\n')) {
          return '"' + v.replace(/"/g, '""') + '"';
        }
        return v;
      })
      .join(',')
  );
  const csv = [header, ...lines].join('\n');
  res.header('Content-Type', 'text/csv');
  res.attachment(name);
  res.send(csv);
}

app.get('/export/activity', (_req, res) => {
  const rows = loadJson(ACTIVITY_PATH, []);
  exportCsvFromArray(rows, res, 'activity.csv');
});

app.get('/export/dismisses', (_req, res) => {
  const rows = loadJson(DISMISSES_PATH, []);
  exportCsvFromArray(rows, res, 'dismisses.csv');
});

app.get('/export/merged', (_req, res) => {
  const activity = loadJson(ACTIVITY_PATH, []).map(r => ({ ...r, type: 'activity' }));
  const dismisses = loadJson(DISMISSES_PATH, []).map(r => ({ ...r, type: 'dismiss' }));
  const rows = [...activity, ...dismisses];
  exportCsvFromArray(rows, res, 'merged.csv');
});

// ---------- API: Suggested Actions (Apply / Undo / Dismiss) ----------

function loadUndo() {
  return loadJson(UNDO_PATH, {});
}
function saveUndo(data) {
  saveJson(UNDO_PATH, data);
}

function mapSuggestedAction({ label, tenant, theme }) {
  if (/spike risk/i.test(label)) {
    return { type: 'ratelimit:raise', label: `Raise RPS for ${theme}`, value: theme, tenant };
  }
  if (/volatile/i.test(label)) {
    return { type: 'runbook:investigate', label: `Investigate volatility in ${theme}`, value: theme, tenant };
  }
  return { type: 'notify:review', label: `Review ${theme}`, value: theme, tenant };
}

// Apply suggested action
app.post('/runbook/suggested_action', (req, res) => {
  try {
    const { label, tenant = 'default', theme } = req.body || {};
    if (!label || !theme) return res.status(400).json({ error: 'label and theme required' });

    const action = mapSuggestedAction({ label, tenant, theme });
    const undo = loadUndo();

    const token = Math.random().toString(16).slice(2) + Math.random().toString(16).slice(2);
    const undoWindowSec = Number(process.env.SUGGESTED_UNDO_SEC || 60);
    const expiresAt = Date.now() + undoWindowSec * 1000;

    undo[token] = { action, expiresAt };
    saveUndo(undo);

    const activity = loadJson(ACTIVITY_PATH, []);
    activity.push({
      timestamp: new Date().toISOString(),
      tenant,
      theme,
      label,
      action: action.label,
      type: action.type,
      undoToken: token,
      undoExpiresAt: new Date(expiresAt).toISOString()
    });
    saveJson(ACTIVITY_PATH, activity);

    res.json({ token, undo_expires_at: new Date(expiresAt).toISOString(), action });
  } catch (e) {
    res.status(500).json({ error: String(e.message || e) });
  }
});

// Undo action
app.post('/runbook/undo_action', (req, res) => {
  try {
    const { token } = req.body || {};
    if (!token) return res.status(400).json({ error: 'token required' });
    const undo = loadUndo();
    const entry = undo[token];
    if (!entry) return res.status(400).json({ error: 'Invalid or expired token' });
    if (Date.now() > entry.expiresAt) {
      delete undo[token];
      saveUndo(undo);
      return res.status(400).json({ error: 'Undo window expired' });
    }

    // here you would actually revert the effect of entry.action
    delete undo[token];
    saveUndo(undo);

    const activity = loadJson(ACTIVITY_PATH, []);
    activity.push({
      timestamp: new Date().toISOString(),
      tenant: entry.action.tenant,
      theme: entry.action.value,
      label: `UNDO ${entry.action.label}`,
      action: 'UNDO',
      type: entry.action.type
    });
    saveJson(ACTIVITY_PATH, activity);

    res.json({ undone: true });
  } catch (e) {
    res.status(500).json({ error: String(e.message || e) });
  }
});

// Dismiss suggestion with reason
app.post('/runbook/dismiss', (req, res) => {
  try {
    const { tenant = 'default', theme, label, reason = 'Other', note = '' } = req.body || {};
    const dismisses = loadJson(DISMISSES_PATH, []);
    const entry = {
      timestamp: new Date().toISOString(),
      tenant,
      theme,
      label,
      reason,
      note
    };
    dismisses.push(entry);
    saveJson(DISMISSES_PATH, dismisses);

    const activity = loadJson(ACTIVITY_PATH, []);
    activity.push({
      timestamp: entry.timestamp,
      tenant,
      theme,
      label: `DISMISS ${label}`,
      action: 'DISMISS',
      type: 'dismiss',
      reason,
      note
    });
    saveJson(ACTIVITY_PATH, activity);

    res.json({ dismissed: true });
  } catch (e) {
    res.status(500).json({ error: String(e.message || e) });
  }
});
